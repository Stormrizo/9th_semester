\begin{AGbox}[8]
\begin{enumerate}[label={{\textcolor{trueblue}{\textbf{Pr.} \arabic*.1}}}, start=3]
  \item Para el sistema de dos partículas libres demostrar que el momento lineal total, $\boldsymbol{P}$,
  y el momento angular total $\boldsymbol{L}$, están dados por:
  \begin{equation}
  \begin{split}
   \boldsymbol{P}  &= m_{1}\dot{\boldsymbol{r}}_{1} + m_{2}\dot{\boldsymbol{r}}_{2}\\
   \mathbf{L} &= \boldsymbol{r}_{1} \times \boldsymbol{p}_{1} + \boldsymbol{r}_{2} \times \boldsymbol{p}_{2}
  \end{split}
  \end{equation}
  respectivamente. Además, demuestre que son vectores constantes en el tiempo.
\end{enumerate}
\end{AGbox} \vspace{0.5cm}

\textcolor{Cinnabar}{\textbf{Solución:}}

Analizamos los \textbf{grados de libertad} del sistema.  
Al tener dos partículas puntuales ($N=2$) que pueden moverse libremente en el espacio tridimensional ($\mathbb{R}^{3}$) y sin restricciones ($k=0$), el número total de grados de libertad es:
\begin{equation}
n = 3N - k = 3(2) - 0 = 6.
\label{eq:grados-libertad}
\end{equation}
Cada partícula requiere tres coordenadas para describir su posición, por lo que el sistema completo posee seis coordenadas independientes:

\[ \boxed{q_{1} = x_{1}}, \quad \boxed{q_{2}=y_{1}}, \quad \boxed{q_{3}=z_{1}}, \quad \boxed{q_{4}=x_{2}}, \quad \boxed{q_{5}=y_{2}}, \quad \boxed{q_{6}=z_{2}} \]

\vspace{0.3cm}

Denotamos los vectores de posición de las partículas $m_1$ y $m_2$ respecto al origen como:
\begin{equation}
\boldsymbol{r}_1 = (x_1,\, y_1,\, z_1), \qquad
\boldsymbol{r}_2 = (x_2,\, y_2,\, z_2).
\label{eq:posiciones}
\end{equation}
Por tanto, las coordenadas generalizadas y sus derivadas temporales son:
\begin{equation}
\boldsymbol{q} = (x_1,\, y_1,\, z_1,\, x_2,\, y_2,\, z_2), 
\qquad
\dot{\boldsymbol{q}} = (\dot{x}_1,\, \dot{y}_1,\, \dot{z}_1,\, \dot{x}_2,\, \dot{y}_2,\, \dot{z}_2)
\label{eq:coordenadas}
\end{equation}

\vspace{0.3cm}

Al tratarse de \textbf{partículas libres}, la energía potencial $U$ es constante y puede fijarse como $U=0$.  
Así, la Lagrangiana depende únicamente de la energía cinética:
\begin{equation}
\begin{split}
L(\boldsymbol{q}, \dot{\boldsymbol{q}}) 
&= \frac{1}{2}m_1\dot{\boldsymbol{r}}_1^{\,2} + \frac{1}{2}m_2\dot{\boldsymbol{r}}_2^{\,2} \\
&= \frac{1}{2}m_1\left(\dot{x}_1^{\,2} + \dot{y}_1^{\,2} + \dot{z}_1^{\,2}\right)
   + \frac{1}{2}m_2\left(\dot{x}_2^{\,2} + \dot{y}_2^{\,2} + \dot{z}_2^{\,2}\right)
\end{split}
\label{eq:lagrangiana-libre}
\end{equation}

\vspace{0.3cm}

Consideremos ahora las dos partículas libres de masas $m_1$ y $m_2$, con vectores de posición $\boldsymbol{r}_1$ y $\boldsymbol{r}_2$.

\begin{figure}[H]
    \centering
	    \includegraphics[width=0.6\textwidth]{1.png}
    \caption{Representación sencilla del sistema de dos partículas libres en el espacio $\mathbb{R}^{3}$ de masas $m_1$ y $m_2$, 
con sus respectivos vectores de posición $\boldsymbol{r}_1$ y $\boldsymbol{r}_2$ medidos desde el origen. 
El vector relativo $\boldsymbol{r}_2 - \boldsymbol{r}_1$ indica la dirección y separación entre ambas partículas.}
    \label{fig:mi_figura2}
\end{figure}

\vspace{0.3cm}

El \textbf{momento lineal individual} de cada partícula se define como:
\begin{equation}
\boldsymbol{p}_i = m_i \dot{\boldsymbol{r}}_i 
= m_i\big(\dot{x}_i\,\hat{\boldsymbol{i}} + \dot{y}_i\,\hat{\boldsymbol{j}} + \dot{z}_i\,\hat{\boldsymbol{k}}\big),
\qquad i = 1,2.
\label{eq:momento-individual}
\end{equation}


A partir de \eqref{eq:momento-individual}, el \textbf{momento lineal total} del sistema resulta de la suma de los momentos individuales de ambas partículas:

\begin{equation}
\boldsymbol{P} = \boldsymbol{p}_1 + \boldsymbol{p}_2 
= m_1\dot{\boldsymbol{r}}_1 + m_2\dot{\boldsymbol{r}}_2.
\label{eq:momento-total}
\end{equation}

De manera explícita, sustituyendo las componentes cartesianas de cada vector 
\(\dot{\boldsymbol{r}}_i = (\dot{x}_i,\, \dot{y}_i,\, \dot{z}_i)\), 
se obtiene:

\[
\boldsymbol{P} 
= m_1(\dot{x}_1\,\hat{\boldsymbol{i}} + \dot{y}_1\,\hat{\boldsymbol{j}} + \dot{z}_1\,\hat{\boldsymbol{k}}) 
 + m_2(\dot{x}_2\,\hat{\boldsymbol{i}} + \dot{y}_2\,\hat{\boldsymbol{j}} + \dot{z}_2\,\hat{\boldsymbol{k}}).
\]

Agrupando las componentes, el \textbf{momento lineal total} puede escribirse como:

\begin{equation}
\boldsymbol{P} = 
\big(m_1\dot{x}_1 + m_2\dot{x}_2\big)\hat{\boldsymbol{i}} \;+\;
\big(m_1\dot{y}_1 + m_2\dot{y}_2\big)\hat{\boldsymbol{j}} \;+\;
\big(m_1\dot{z}_1 + m_2\dot{z}_2\big)\hat{\boldsymbol{k}}.
\label{eq:momento-total-componentes}
\end{equation}

Por tanto, las \textbf{componentes escalares} del momento lineal total del sistema son:
\[
\boxed{
\begin{aligned}
P_x &= m_1\dot{x}_1 + m_2\dot{x}_2, \\[2pt]
P_y &= m_1\dot{y}_1 + m_2\dot{y}_2, \\[2pt]
P_z &= m_1\dot{z}_1 + m_2\dot{z}_2.
\end{aligned}
}
\]

Cada componente representa la cantidad de movimiento total en su respectiva dirección cartesiana.
En conjunto, el vector \(\boldsymbol{P} = (P_x, P_y, P_z)\) 
describe el impulso total del sistema de dos partículas en el espacio tridimensional.


Para estudiar la evolución temporal de $\boldsymbol{P}$, partimos de \eqref{eq:momento-total} y derivamos con respecto al tiempo:
\begin{equation}
\begin{split}
\dot{\boldsymbol{P}}
&= \frac{d}{dt}\Big(\boldsymbol{p}_1 + \boldsymbol{p}_2\Big)
 = \dot{\boldsymbol{p}}_1 + \dot{\boldsymbol{p}}_2.
\end{split}
\label{eq:derivada-p}
\end{equation}

A partir de la definición del momento lineal individual \eqref{eq:momento-individual},
su derivada temporal componente a componente es:

\begin{equation}
\begin{split}
\dot{\boldsymbol{p}}_i 
&= \frac{d}{dt}(m_i \dot{\boldsymbol{r}}_i) = m_i \ddot{\boldsymbol{r}}_i \\[4pt]
&\Rightarrow 
\begin{cases}
\dot{p}_{ix} = m_i \ddot{x}_i, \\[4pt]
\dot{p}_{iy} = m_i \ddot{y}_i, \\[4pt]
\dot{p}_{iz} = m_i \ddot{z}_i,
\end{cases}
\end{split}
\label{eq:derivada-pi-escalar}
\end{equation}

lo cual, por la \textbf{segunda ley de Newton}, equivale a escribir:

\begin{equation}
\begin{cases}
F_{ix} = m_i \ddot{x}_i, \\[4pt]
F_{iy} = m_i \ddot{y}_i, \\[4pt]
F_{iz} = m_i \ddot{z}_i,
\end{cases}
\qquad \text{o bien, en forma vectorial:} \quad 
\boldsymbol{F}_i = m_i \ddot{\boldsymbol{r}}_i.
\label{eq:newton-escalar}
\end{equation}

De \eqref{eq:derivada-pi-escalar} y \eqref{eq:newton-escalar} se obtiene la relación directa entre la variación temporal del momento y la fuerza aplicada:

\[
\dot{p}_{ix} = F_{ix}, \qquad 
\dot{p}_{iy} = F_{iy}, \qquad 
\dot{p}_{iz} = F_{iz},
\]
oen forma vectorial
\(\dot{\boldsymbol{p}}_i = \boldsymbol{F}_i\).


El \textbf{momento lineal total} del sistema en cada dirección cartesiana se escribe entonces como:

\begin{equation}
\begin{cases}
P_x = p_{1x} + p_{2x} = m_1\dot{x}_1 + m_2\dot{x}_2, \\[4pt]
P_y = p_{1y} + p_{2y} = m_1\dot{y}_1 + m_2\dot{y}_2, \\[4pt]
P_z = p_{1z} + p_{2z} = m_1\dot{z}_1 + m_2\dot{z}_2.
\end{cases}
\label{eq:P-escalar}
\end{equation}

Derivando con respecto al tiempo, se obtiene la variación temporal del momento lineal total:

\begin{equation}
\begin{cases}
\dot{P}_x = \dot{p}_{1x} + \dot{p}_{2x} = F_{1x} + F_{2x}, \\[4pt]
\dot{P}_y = \dot{p}_{1y} + \dot{p}_{2y} = F_{1y} + F_{2y}, \\[4pt]
\dot{P}_z = \dot{p}_{1z} + \dot{p}_{2z} = F_{1z} + F_{2z}.
\end{cases}
\label{eq:variacion-momento-escalar}
\end{equation}

En forma vectorial, esto corresponde a:
\[
\dot{\boldsymbol{P}} = \dot{\boldsymbol{p}}_1 + \dot{\boldsymbol{p}}_2 = \boldsymbol{F}_1 + \boldsymbol{F}_2.
\]

Si el sistema está aislado, sólo existen las fuerzas \textbf{internas} entre las partículas:
\(\boldsymbol{F}_{1\leftarrow2}\) (sobre 1 por 2) y \(\boldsymbol{F}_{2\leftarrow1}\) (sobre 2 por 1).  
Por la \textbf{tercera ley de Newton}:

\[
\boldsymbol{F}_{1\leftarrow2} = -\,\boldsymbol{F}_{2\leftarrow1}.
\]

Entonces, cada componente cumple:

\[
\begin{cases}
F_{1\leftarrow2,\,x} + F_{2\leftarrow1,\,x} = 0, \\[4pt]
F_{1\leftarrow2,\,y} + F_{2\leftarrow1,\,y} = 0, \\[4pt]
F_{1\leftarrow2,\,z} + F_{2\leftarrow1,\,z} = 0,
\end{cases}
\]

lo cual implica directamente que:

\begin{equation}
\dot{P}_x = \dot{P}_y = \dot{P}_z = 0.
\label{eq:conservacion-P-escalar}
\end{equation}

Es decir que si la derivada temporal de una cantidad física es cero, significa que dicha cantidad no cambia con el tiempo, es decir, se \textbf{conserva}.  
Matemáticamente,
\[
\frac{d}{dt}(\text{constante}) = 0,
\]
y físicamente esto se traduce en que el momento lineal total del sistema permanece invariable durante la evolución del movimiento.

Por tanto, el momento lineal total se conserva \emph{en cada dirección independiente del espacio}, lo que expresa que en un sistema aislado ninguna de las componentes \(P_x\), \(P_y\) o \(P_z\) varía en el tiempo:
\[
\boxed{
P_x = \text{cte.}, \quad
P_y = \text{cte.}, \quad
P_z = \text{cte.}
}
\]


\[
\boxed{
\boldsymbol{P} = \text{cte.} \quad \Rightarrow \quad 
(P_x, P_y, P_z) = \text{constantes.}
}
\]


% =========================================================
% Momento angular total: desarrollo con matrices (determinantes)
% =========================================================

Estamos ahora en la parte de demostrar la siguiente ecuación del momento angular total:
\begin{equation}
\boldsymbol{L}=\boldsymbol{r}_1\times\boldsymbol{p}_1+\boldsymbol{r}_2\times\boldsymbol{p}_2
\label{eq:def-L-matrices}
\end{equation}

% =========================================================
% Momento angular total: determinantes, componentes y derivadas
% =========================================================

\textcolor{orange(colorwheel)}{\textbf{Producto cruz mediante determinantes (expansión por cofactores).}} 

Para cada partícula:

\textcolor{Cerulean}{\textbf{Partícula de masa $m_{1}$}}:
\[
\boldsymbol{r}_1\times\boldsymbol{p}_1=
\begin{vmatrix}
\hat{\boldsymbol{i}} & \hat{\boldsymbol{j}} & \hat{\boldsymbol{k}}\\
x_1 & y_1 & z_1\\
p_{1x} & p_{1y} & p_{1z}
\end{vmatrix}
=
(y_1p_{1z}-z_1p_{1y})\,\hat{\boldsymbol{i}}
-(x_1p_{1z}-z_1p_{1x})\,\hat{\boldsymbol{j}}
+(x_1p_{1y}-y_1p_{1x})\,\hat{\boldsymbol{k}}
\]

\textcolor{pakistangreen}{\textbf{Partícula de masa $m_{2}$}}:
\[
\boldsymbol{r}_2\times\boldsymbol{p}_2=
\begin{vmatrix}
\hat{\boldsymbol{i}} & \hat{\boldsymbol{j}} & \hat{\boldsymbol{k}}\\
x_2 & y_2 & z_2\\
p_{2x} & p_{2y} & p_{2z}
\end{vmatrix}
=
(y_2p_{2z}-z_2p_{2y})\,\hat{\boldsymbol{i}}
-(x_2p_{2z}-z_2p_{2x})\,\hat{\boldsymbol{j}}
+(x_2p_{2y}-y_2p_{2x})\,\hat{\boldsymbol{k}}
\]

\textcolor{Apple Green}{\textbf{Componentes escalares de $\boldsymbol L$}} (sumando ambas contribuciones):
\[
\boxed{
\begin{aligned}
L_x &= (y_1p_{1z}-z_1p_{1y})+(y_2p_{2z}-z_2p_{2y})\\[2pt]
L_y &= (z_1p_{1x}-x_1p_{1z})+(z_2p_{2x}-x_2p_{2z})\\[2pt]
L_z &= (x_1p_{1y}-y_1p_{1x})+(x_2p_{2y}-y_2p_{2x})
\end{aligned}
}
\]

% ------------------ Derivadas: paso explícito + cancelaciones justificadas ------------------

\textcolor{vividviolet}{\textbf{Derivadas temporales.}} 
Derivamos cada componente usando la regla del producto:
\[
\begin{cases}
\dot{L}_x = (\dot{y}_1p_{1z}+y_1\dot{p}_{1z})-(\dot{z}_1p_{1y}+z_1\dot{p}_{1y})
           + (\dot{y}_2p_{2z}+y_2\dot{p}_{2z})-(\dot{z}_2p_{2y}+z_2\dot{p}_{2y}), \\[6pt]
\dot{L}_y = (\dot{z}_1p_{1x}+z_1\dot{p}_{1x})-(\dot{x}_1p_{1z}+x_1\dot{p}_{1z})
           + (\dot{z}_2p_{2x}+z_2\dot{p}_{2x})-(\dot{x}_2p_{2z}+x_2\dot{p}_{2z}), \\[6pt]
\dot{L}_z = (\dot{x}_1p_{1y}+x_1\dot{p}_{1y})-(\dot{y}_1p_{1x}+y_1\dot{p}_{1x})
           + (\dot{x}_2p_{2y}+x_2\dot{p}_{2y})-(\dot{y}_2p_{2x}+y_2\dot{p}_{2x})
\end{cases}
\]

Los términos del tipo
\(\dot{y}_i p_{iz} - \dot{z}_i p_{iy}\), \(\dot{z}_i p_{ix} - \dot{x}_i p_{iz}\), 
\(\dot{x}_i p_{iy} - \dot{y}_i p_{ix}\) son \emph{exactamente} las componentes del producto cruz 
\(\dot{\boldsymbol r}_i \times \boldsymbol p_i\):
\[
(\dot{\boldsymbol r}_i \times \boldsymbol p_i)_x=\dot y_i p_{iz}-\dot z_i p_{iy},\quad
(\dot{\boldsymbol r}_i \times \boldsymbol p_i)_y=\dot z_i p_{ix}-\dot x_i p_{iz},\quad
(\dot{\boldsymbol r}_i \times \boldsymbol p_i)_z=\dot x_i p_{iy}-\dot y_i p_{ix}.
\]

Ahora como \(\boldsymbol p_i = m_i \dot{\boldsymbol r}_i\), velocidad y momento son paralelos. 
Por el determinante tenemos:
\[
\dot{\boldsymbol r}_i \times \boldsymbol p_i =
\begin{vmatrix}
\hat{\boldsymbol i} & \hat{\boldsymbol j} & \hat{\boldsymbol k}\\
\dot x_i & \dot y_i & \dot z_i\\
p_{ix} & p_{iy} & p_{iz}
\end{vmatrix}
=
(\dot y_i p_{iz}-\dot z_i p_{iy})\,\hat{\boldsymbol i}
+(\dot z_i p_{ix}-\dot x_i p_{iz})\,\hat{\boldsymbol j}
+(\dot x_i p_{iy}-\dot y_i p_{ix})\,\hat{\boldsymbol k}.
\]

\textbf{Componente $x$}

Sustituyendo \(p_{i\alpha}=m_i \dot r_{i\alpha}\) (con \(\alpha=x,y,z\)), se tiene lo siguiente:
\[
\begin{aligned}
(\dot{\boldsymbol r}_i \times \boldsymbol p_i)_x
&= \dot y_i (m_i \dot z_i) - \dot z_i (m_i \dot y_i) \\[4pt]
&= m_i\,\underbrace{(\dot y_i \dot z_i - \dot z_i \dot y_i)}_{\substack{\text{t\'erminos iguales}\\ \text{i.e. } ab - ba=0}} = 0\\
&= m_i(0) = 0.
\end{aligned}
\]

Así para las expresiones de \(\dot{L}_x,\dot{L}_y,\dot{L}_z\) se cancelan \emph{todos} los términos de la forma 
\((\dot{\boldsymbol r}_i \times \boldsymbol p_i)_\alpha\). 
Sobreviven únicamente los términos con \(\dot{\boldsymbol p}_i\), que, por la segunda ley de Newton, 
son fuerzas: \(\dot{\boldsymbol p}_i=\boldsymbol F_i\). 
Así, reagrupando por componentes,
\[
\boxed{
\begin{aligned}
\dot{L}_x &= (y_1F_{1z}-z_1F_{1y})+(y_2F_{2z}-z_2F_{2y})\\[2pt]
\dot{L}_y &= (z_1F_{1x}-x_1F_{1z})+(z_2F_{2x}-x_2F_{2z})\\[2pt]
\dot{L}_z &= (x_1F_{1y}-y_1F_{1x})+(x_2F_{2y}-y_2F_{2x})
\end{aligned}
}
\]

observamos que cada conjunto de términos tiene la misma estructura que las componentes de un producto vectorial del tipo \(\boldsymbol{r}_i\times\boldsymbol{F}_i\).  
Para verlo de manera explícita, consideremos el desarrollo del producto cruz mediante \textcolor{orange(colorwheel)}{\textbf{expansión por cofactores}}:

\[
\boldsymbol{r}_i\times\boldsymbol{F}_i =
\begin{vmatrix}
\hat{\boldsymbol{i}} & \hat{\boldsymbol{j}} & \hat{\boldsymbol{k}}\\
x_i & y_i & z_i\\
F_{ix} & F_{iy} & F_{iz}
\end{vmatrix}
= (y_iF_{iz}-z_iF_{iy})\,\hat{\boldsymbol{i}}
+ (z_iF_{ix}-x_iF_{iz})\,\hat{\boldsymbol{j}}
+ (x_iF_{iy}-y_iF_{ix})\,\hat{\boldsymbol{k}}.
\]

Cada una de estas componentes coincide exactamente con los términos de las ecuaciones anteriores:  
la primera línea con la componente \(x\), la segunda con la componente \(y\) y la tercera con la componente \(z\).  
Por tanto, podemos reescribir el conjunto de tres ecuaciones escalares en una sola ecuación vectorial.
\[
\boxed{\ \dot{\boldsymbol{L}} = (\boldsymbol{r}_1\times\boldsymbol{F}_1) + (\boldsymbol{r}_2\times\boldsymbol{F}_2)\ }\!,
\]

lo que muestra que la variación temporal del momento angular total es igual a la suma de los \textbf{torques} (momentos de fuerza) ejercidos sobre cada partícula del sistema.


% ------------------ Caso de fuerzas internas centrales ------------------

\textcolor{trueblue}{\textbf{Caso de fuerzas internas centrales.}}

En un sistema aislado, las únicas fuerzas que actúan son \textbf{internas}, es decir, aquellas que las partículas ejercen entre sí.  
Sea \(\boldsymbol{F}_{1\leftarrow 2}\) la fuerza que la partícula 2 ejerce sobre la partícula 1, y \(\boldsymbol{F}_{2\leftarrow 1}=-\boldsymbol{F}_{1\leftarrow 2}\) su correspondiente fuerza de reacción, de acuerdo con la \textbf{tercera ley de Newton}.  

Si además dichas fuerzas son \emph{centrales}, esto es, actúan a lo largo de la línea que une a ambas partículas,
\[
\boldsymbol{F}_{1\leftarrow 2} \parallel \boldsymbol{r} = \boldsymbol{r}_2 - \boldsymbol{r}_1,
\]
entonces el torque interno total del sistema se anula.  
En efecto, partiendo de la relación general
\[
\dot{\boldsymbol{L}} = \boldsymbol{r}_1 \times \boldsymbol{F}_1 + \boldsymbol{r}_2 \times \boldsymbol{F}_2,
\]
y sustituyendo \(\boldsymbol{F}_1 = \boldsymbol{F}_{1\leftarrow 2}\), \(\boldsymbol{F}_2 = \boldsymbol{F}_{2\leftarrow 1} = -\boldsymbol{F}_{1\leftarrow 2}\), se obtiene
\begin{equation}
\begin{split}
\dot{\boldsymbol{L}} 
&= \boldsymbol{r}_1 \times \boldsymbol{F}_{1\leftarrow 2} 
   + \boldsymbol{r}_2 \times \boldsymbol{F}_{2\leftarrow 1} 
   \quad \text{(definición general del torque total)} \\[6pt]
&= \boldsymbol{r}_1 \times \boldsymbol{F}_{1\leftarrow 2} 
   + \boldsymbol{r}_2 \times (-\boldsymbol{F}_{1\leftarrow 2})
   \quad \text{(tercera ley: } \boldsymbol{F}_{2\leftarrow 1}=-\boldsymbol{F}_{1\leftarrow 2}\text{)} \\[6pt]
&= \boldsymbol{r}_1 \times \boldsymbol{F}_{1\leftarrow 2} 
   - \boldsymbol{r}_2 \times \boldsymbol{F}_{1\leftarrow 2}
   \quad \text{(linealidad en el segundo argumento: } \boldsymbol{a}\times(\alpha\,\boldsymbol{b})=\alpha\,(\boldsymbol{a}\times\boldsymbol{b})\text{)} \\[6pt]
&= (\boldsymbol{r}_1 - \boldsymbol{r}_2)\times \boldsymbol{F}_{1\leftarrow 2}
   \quad \text{(distributividad: } \boldsymbol{a}\times\boldsymbol{c} - \boldsymbol{b}\times\boldsymbol{c}
   = (\boldsymbol{a}-\boldsymbol{b})\times\boldsymbol{c}\text{)} \\[6pt]
&= \boldsymbol{r}\times \boldsymbol{F}_{1\leftarrow 2}
   \quad \text{(definición: } \boldsymbol{r}=\boldsymbol{r}_1-\boldsymbol{r}_2\text{)}
\end{split}
\label{eq:torque-interno}
\end{equation}


Como el vector de fuerza \(\boldsymbol{F}_{1\leftarrow 2}\) es paralelo al vector de posición relativa \(\boldsymbol{r}\),
su producto vectorial es nulo:
\[
\boxed{\ \dot{\boldsymbol{L}} = \boldsymbol{0}\ }
\]
Por lo tanto, el \textbf{momento angular total del sistema se conserva}:
\[
\boxed{\ \boldsymbol{L} = \text{constante}\ }
\]

\vspace{0.5cm}

\textcolor{trueblue}{\textbf{Conservación del momento lineal.}}

De manera análoga, para el momento lineal total se cumple
\[
\dot{\boldsymbol{P}} = \boldsymbol{F}_1 + \boldsymbol{F}_2.
\]
Si sólo actúan fuerzas internas, entonces
\(\boldsymbol{F}_{1\leftarrow 2} = -\boldsymbol{F}_{2\leftarrow 1}\), y por tanto
\[
\boxed{\ \dot{\boldsymbol{P}} = \boldsymbol{0} \quad \Rightarrow \quad \boldsymbol{P} = \text{constante}\ }.
\]

\textcolor{Cinnabar}{\textbf{Solución:}}

La ecuación dada
\[
y = -x\cot\bar{\phi} + \sqrt{\frac{l^2}{2mE}}\,\csc\bar{\phi}
\]
corresponde al caso libre del \textit{problema de Kepler}, es decir, al movimiento de una partícula con energía total \(E\) y momento angular \(l\) constantes bajo una fuerza central nula. 

Podemos realizar un cambio de variable tomando como la variable $a$ al siguiente término de la ecuación que se nos da en el enunciado, tenemos:

\begin{equation}
a \;\equiv\; \sqrt{\frac{l^2}{2mE}} \;=\; \frac{l}{\sqrt{2mE}}
\label{eq:def-a}
\end{equation}

De la definición \eqref{eq:def-a}, la familia de rectas del enunciado puede escribirse de forma compacta como

\begin{equation}
\begin{split}
{\color{onyx}{y}} 
&= -\,{\color{AppleGreen}{x}}\,{\color{orange}{\cot\bar{\phi}}}
  + {\color{Cerulean}{\sqrt{\tfrac{l^{2}}{2mE}}}}\,
    {\color{Cinnabar}{\csc\bar{\phi}}}\\[4pt]
&= -x{\color{orange}{\cot\bar{\phi}}}
  + {\color{Cerulean}{a}}\,{\color{Cinnabar}{\csc\bar{\phi}}}
  \quad \text{tomando}\quad
  \boxed{{\color{Cerulean}{a=\sqrt{\tfrac{l^{2}}{2mE}}}}}\\[4pt]
&= -x\!\left({\color{orange}{\frac{\cos\bar{\phi}}{\sen\bar{\phi}}}}\right)
  + a\!\left({\color{Cinnabar}{\frac{1}{\sen\bar{\phi}}}}\right)
\end{split}\label{eq:recta-compacta}
\end{equation}

Para visualizar mejor la geometría de la trayectoria, tomamos la ecuación 
en forma compacta~\eqref{eq:recta-compacta} y la multiplicamos por 
\({\color{Cinnabar}{\sen\bar{\phi}}}\). Este paso elimina los denominadores 
y evita las \textbf{indeterminaciones} que aparecen cuando
\(\sen\bar{\phi}=0\) (\(\bar{\phi}=0,\pi,2\pi\)).

Así, al “barrer” el ángulo \(\bar{\phi}\) en el intervalo 
\([0,2\pi]\), obtenemos una descripción continua y simétrica de todas las 
rectas tangentes al círculo \(r=a\), sin asíntotas ni discontinuidades 
en su representación. De esta forma, la ecuación adopta una versión más 
limpia y regular:



\begin{equation}
\begin{split}
{\color{onyx}{y}}
&= -x{\color{orange}{\cot\bar{\phi}}}
   + {\color{Cerulean}{a}}\,{\color{Cinnabar}{\csc\bar{\phi}}}
   \quad \text{(ecuación original de la recta)} \\[6pt]
{\color{Cinnabar}{\sen\bar{\phi}}}\,y
&= -x{\color{orange}{\cos\bar{\phi}}}
   + {\color{Cerulean}{a}}
   \quad \text{(multiplicamos por $\sen\bar{\phi}$ para eliminar denominadores)} \\[6pt]
{\color{AppleGreen}{x}}{\color{orange}{\cos\bar{\phi}}}
+{\color{AppleGreen}{y}}{\color{Cinnabar}{\sen\bar{\phi}}}
&={\color{Cerulean}{a}}
\quad \text{(reordenando términos, obtenemos la forma normal)}
\end{split}
\label{eq:recta-normal}
\end{equation}


Podemos notar que el parámetro 

\[
\boxed{{\color{Cerulean}{a = \frac{l}{\sqrt{2mE}}}}}
\]

permanece constante a lo largo de toda la familia de rectas. 
Esto se debe a que 
\({\color{Cerulean}{\boldsymbol{a}}}\) depende exclusivamente del momento 
angular \({\color{Cerulean}{\boldsymbol{l}}}\) y de la energía total 
\({\color{Cerulean}{\boldsymbol{E}}}\), ambas 
\emph{magnitudes invariantes del movimiento} en un sistema gobernado por 
una fuerza central conservativa. 
En otras palabras, si el sistema conserva \(\boldsymbol{l}\) y \(\boldsymbol{E}\), entonces todas 
las rectas comparten la misma distancia fija 
\({\color{Cerulean}{a}}\) respecto al origen, lo que define la 
envolvente circular de radio \(r = {\color{Cerulean}{a}}\).\\[6pt]

Para ilustrar este comportamiento, podemos asignar 
valores numéricos concretos a las magnitudes físicas. 
Por ejemplo, si tomamos una masa \(\boxed{\boldsymbol{m = 1}}\) (en unidades arbitrarias), una 
energía \(\boxed{\color{Cerulean}\boldsymbol{E = 2}}\) y un momento angular \(\boxed{\color{Cerulean}\boldsymbol{l = 3}}\), obtenemos: 

\[
{\color{Cerulean}{a = \frac{l}{\sqrt{2mE}} = \frac{3}{\sqrt{4}} = 1\text{.}5}}
\]

De este modo, el valor 
\({\color{Cerulean}{a = 1\text{.}5}}\) 
corresponde al radio común a toda la familia de rectas generadas por la ecuación~\eqref{eq:recta-normal}.

\subsection*{Cambio a variable \(u(\phi)=1/r\) (ecuación de Binet)}
Relación:
\[
\dot r = \frac{dr}{dt} = \frac{dr}{d\phi}\dot\phi = \frac{dr}{d\phi}\frac{\ell}{\mu r^2}.
\]
Con \(u=1/r\), \(dr/d\phi = -u^{-2} du/d\phi\) y
\[
\dot r = -\frac{\ell}{\mu}\frac{du}{d\phi}.
\]
Sustituyendo en \(E\) y manipulando,
\[
\Bigl(\frac{du}{d\phi}\Bigr)^2 + u^2 - \frac{2\mu\alpha}{\ell^2}u = \frac{2\mu E}{\ell^2}.
\]
Derivando esta identidad respecto. a \(\phi\) y reordenando (o usando derivación Newtoniana) se obtiene la ecuación lineal:
\[
\frac{d^2 u}{d\phi^2} + u = \frac{\mu\alpha}{\ell^2}.
\]
Solución general:
\[
u(\phi) = \frac{\mu\alpha}{\ell^2}\Bigl[1 + e\cos(\phi-\phi_0)\Bigr],
\]
o en la forma clásica de la cónica:
\[
\boxed{\displaystyle r(\phi) = \frac{p}{1+e\cos(\phi-\phi_0)},\qquad p=\frac{\ell^2}{\mu\alpha}.}
\]

\subsection*{Relación entre \(e\) y \(E\)}
Sustituyendo la solución en la identidad previa se llega a:
\[
A^2 e^2 = A^2 + \frac{2\mu E}{\ell^2},
\qquad A=\frac{\mu\alpha}{\ell^2}.
\]
Despejando,
\[
\boxed{e^2 = 1 + \frac{2E\ell^2}{\mu\alpha^2}.}
\]
Por tanto:
\begin{itemize}
    \item \(E<0 \Rightarrow e<1\) (elipse),
    \item \(E=0 \Rightarrow e=1\) (parábola),
    \item \(E>0 \Rightarrow e>1\) (hipérbola).
\end{itemize}

\subsection*{Parámetros orbitales}
Semieje mayor \(a\) (para elipses, \(E<0\)):
\[
\frac{E}{\mu} = -\frac{\alpha}{2a} \quad\Rightarrow\quad a = -\frac{\mu\alpha}{2E}.
\]
Relaciones:
\[
p = a(1-e^2),\qquad r_{\min} = a(1-e),\qquad r_{\max}=a(1+e).
\]

\subsection*{Determinación de \(e\) y fase desde condiciones iniciales}
Dado \(r_0\) y \(\dot r_0\) en \(\phi=\phi_{\mathrm{init}}\), con \(u_0=1/r_0\),
\[
\begin{cases}
u_0 = A\bigl(1 + e\cos\delta\bigr),\\[4pt]
\dot r_0 = \dfrac{\ell}{\mu}A e\sin\delta,
\end{cases}
\quad \delta = \phi_{\mathrm{init}} - \phi_0.
\]
De aquí:
\[
e^2 = \Bigl(\frac{u_0}{A}-1\Bigr)^2 + \Bigl(\frac{\mu\dot r_0}{\ell A}\Bigr)^2,
\quad
\delta = \operatorname{atan2}\!\Bigl(\frac{\mu\dot r_0}{\ell A},\,\frac{u_0}{A}-1\Bigr).
\]
