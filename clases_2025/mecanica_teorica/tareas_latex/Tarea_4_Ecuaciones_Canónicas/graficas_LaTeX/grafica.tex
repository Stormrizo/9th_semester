\documentclass[
  tikz,
  border=5pt,
  convert={density=600,outext=.png}
]{standalone}


\usepackage{pgfplots}
\pgfplotsset{compat=1.18}

% Paleta de colores
\input{colores.tex}

\begin{document}

\begin{tikzpicture}
\begin{axis}[
    view={60}{25},
    xlabel={$x$},
    ylabel={$p$},
    zlabel={$t$},
    axis lines=middle,
    width=12cm,
    height=8cm,
    xmin=-8, xmax=8,   % porque x = (p0/2)t, con |p0|=4 y |t|=4
    ymin=-4, ymax=4,
    zmin=-4, zmax=4,
    legend style={
        font=\footnotesize,
        at={(0.02,0.98)},
        anchor=north west
    },
    legend cell align=left,
    legend columns=2,
]

  % m = 2, x0 = 0, t \in [-4,4]
  % x(t) = (p0/2) t,  p(t) = p0,  t = t

  % p0 = -4
  \addplot3[
    very thick,
    color=Apple Green,
    domain=-4:4,
    samples=30,
    variable=\t
  ] ({-2*\t}, {-4}, {\t});
  \addlegendentry{$p_{0}=-4$}

  % p0 = -3
  \addplot3[
    very thick,
    color=Surfie Green,
    domain=-4:4,
    samples=30,
    variable=\t
  ] ({-1.5*\t}, {-3}, {\t});
  \addlegendentry{$p_{0}=-3$}

  % p0 = -2
  \addplot3[
    very thick,
    color=brightturquoise,
    domain=-4:4,
    samples=30,
    variable=\t
  ] ({-1*\t}, {-2}, {\t});
  \addlegendentry{$p_{0}=-2$}

  % p0 = -1
  \addplot3[
    very thick,
    color=Medium Aquamarine,
    domain=-4:4,
    samples=30,
    variable=\t
  ] ({-0.5*\t}, {-1}, {\t});
  \addlegendentry{$p_{0}=-1$}

  % p0 = 0
  \addplot3[
    very thick,
    color=Cinnabar,
    domain=-4:4,
    samples=30,
    variable=\t
  ] ({0}, {0}, {\t});
  \addlegendentry{$p_{0}=0$}

  % p0 = 1
  \addplot3[
    very thick,
    color=Tarawera,
    domain=-4:4,
    samples=30,
    variable=\t
  ] ({0.5*\t}, {1}, {\t});
  \addlegendentry{$p_{0}=1$}

  % p0 = 2
  \addplot3[
    very thick,
    color=Cerulean,
    domain=-4:4,
    samples=30,
    variable=\t
  ] ({1*\t}, {2}, {\t});
  \addlegendentry{$p_{0}=2$}

  % p0 = 3
  \addplot3[
    very thick,
    color=trueblue,
    domain=-4:4,
    samples=30,
    variable=\t
  ] ({1.5*\t}, {3}, {\t});
  \addlegendentry{$p_{0}=3$}

  % p0 = 4
  \addplot3[
    very thick,
    color=palatinateblue,
    domain=-4:4,
    samples=30,
    variable=\t
  ] ({2*\t}, {4}, {\t});
  \addlegendentry{$p_{0}=4$}

\end{axis}
\end{tikzpicture}

\end{document}
