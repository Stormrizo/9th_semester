\section{Desarrollo}


Sea $n(x)$ una función periódica de periodo $a$ y real (físicamente, densidad). Su serie de Fourier compleja se escribe como
\begin{equation*}
\begin{split}
n(x) &= \sum_{p\in\mathbb{Z}} n_{p}\,e^{\,i \tfrac{2\pi p}{a}x}.
\end{split}
\end{equation*}
La familia $\{e^{\,i \tfrac{2\pi p}{a}x}\}_{p\in\mathbb{Z}}$ forma una base ortogonal de $L^{2}[0,a]$ con el producto interno
$\langle f,g\rangle = \tfrac{1}{a}\!\int_{0}^{a} f(x)\,g^{*}(x)\,dx$. La \textbf{ortogonalidad} se codifica en
\begin{equation*}
\begin{split}
\frac{1}{a}\int_{0}^{a} e^{\,i \tfrac{2\pi (p-q)}{a}x}\,dx &= \delta_{p,q},
\end{split}
\end{equation*}
que equivale a la identidad (sin normalizar)
\begin{equation*}
\begin{split}
\int_{0}^{a} e^{\,i \tfrac{2\pi (p-q)}{a}x}\,dx &= a\,\delta_{p,q}.
\end{split}
\end{equation*}

\textbf{¿Por qué aparece la integral?} Porque los coeficientes se obtienen \emph{proyectando} $n(x)$ sobre los vectores base usando ese producto interno (la “medida” de cuánto de $e^{\,i 2\pi p x/a}$ hay dentro de $n$). De ahí sale la fórmula de proyección:
\begin{equation*}
\begin{split}
n_p &= \frac{1}{a}\int_{0}^{a} n(x)\,e^{-\,i \tfrac{2\pi p}{a}x}\,dx.
\end{split}
\end{equation*}

Con esto, la \textbf{realidad} de $n(x)$ (esto es, $n(x)=n^{*}(x)$) impone una \emph{simetría de conjugación} sobre los $n_p$. Veámoslo por dos rutas complementarias.

\medskip
\textbf{\textit{Vía proyección (con integral).}} Partimos de la definición anterior y conjugamos:
\begin{equation*}
\begin{split}
n_{-p}^{*} &= \left(\frac{1}{a}\int_{0}^{a} n(x)\,e^{-\,i \tfrac{2\pi (-p)}{a}x}\,dx\right)^{\!*}
           = \frac{1}{a}\int_{0}^{a} n^{*}(x)\,e^{-\,i \tfrac{2\pi p}{a}x}\,dx.
\end{split}
\end{equation*}
Como $n(x)=n^{*}(x)$, concluimos
\begin{equation*}
\begin{split}
n_{-p}^{*} &= \frac{1}{a}\int_{0}^{a} n(x)\,e^{-\,i \tfrac{2\pi p}{a}x}\,dx \;=\; n_{p}.
\end{split}
\end{equation*}

\medskip
\textbf{\textit{Vía sumatoria (sin integral).}} Conjugamos la serie:
\begin{equation*}
\begin{split}
n^{*}(x) &= \left(\sum_{p} n_p\,e^{\,i \tfrac{2\pi p}{a}x}\right)^{*}
         = \sum_{p} n_p^{*}\,e^{-\,i \tfrac{2\pi p}{a}x}
         = \sum_{q} n_{-q}^{*}\,e^{\,i \tfrac{2\pi q}{a}x}.
\end{split}
\end{equation*}
La igualdad $n(x)=n^{*}(x)$ junto con la \emph{independencia lineal} de los exponentes (o, equivalentemente, su ortogonalidad) fuerza la igualdad coeficiente a coeficiente:
\begin{equation*}
\begin{split}
n_q &= n_{-q}^{*}\qquad(\forall\,q\in\mathbb{Z}).
\end{split}
\end{equation*}
Ambas vías llevan al mismo resultado clave:
\begin{equation*}
\begin{split}
\boxed{\,n_{-p}^{*}=n_{p}\,}.
\end{split}
\end{equation*}

A partir de aquí se obtienen consecuencias útiles. En primer lugar, el coeficiente promedio es real:
\begin{equation*}
\begin{split}
n_{0} &= \frac{1}{a}\int_{0}^{a} n(x)\,dx \in \mathbb{R}.
\end{split}
\end{equation*}
En segundo lugar, separando partes real e imaginaria para $p>0$:
\begin{equation*}
\begin{split}
\Re(n_{-p}) &= \Re(n_{p}), \qquad \Im(n_{-p}) \;=\; -\,\Im(n_{p}).
\end{split}
\end{equation*}
Finalmente, la serie puede escribirse de forma \emph{explícitamente real}:
\begin{equation*}
\begin{split}
n(x) &= n_{0} \;+\; 2\sum_{p=1}^{\infty}\Re\!\Big[n_{p}\,e^{\,i \tfrac{2\pi p}{a}x}\Big] \\[1mm]
     &= n_{0} \;+\; 2\sum_{p=1}^{\infty}\Big(\Re(n_p)\cos\tfrac{2\pi p}{a}x \;-\; \Im(n_p)\sin\tfrac{2\pi p}{a}x\Big).
\end{split}
\end{equation*}

\textbf{Comentario técnico.} La “igualdad coeficiente a coeficiente” puede justificarse de dos maneras: (i) por \emph{unicidad} de la expansión en una base (si $\sum c_p e^{i 2\pi p x/a}=0$ en $[0,a]$, entonces $c_p=0$ para todo $p$), o (ii) multiplicando por $e^{-i 2\pi q x/a}$ e \emph{integrando} en $[0,a]$ para usar $a\,\delta_{p,q}$, que “proyecta” y aísla cada coeficiente. Esto explica por qué la integral aparece de forma natural en el desarrollo: es la operación de proyección asociada al producto interno en $L^{2}[0,a]$.
