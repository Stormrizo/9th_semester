\subsection{Ejercicio de clase: Compute the total energy of a charged dielectric sphere that produces an electric field in the outside and inside regions}

\begin{figure}[H]
    \centering
    \includegraphics[width=0.8\textwidth]{3.png}
    \caption{Esfera dieléctrica de radio $R$ con densidad de carga volumétrica uniforme \(\rho\), inmersa en un medio lineal}
    \label{fig:esfera_dielec}
\end{figure}

\noindent\textbf{Hipótesis y datos}
\begin{itemize}
  \item Esfera de radio \(R\) con densidad de carga \(\rho=\text{cte}\) en su volumen
  \item Materiales lineales, isótropos y homogéneos: \(\vec D=\varepsilon \vec E\)
  \item Permitividad dentro de la esfera \(\varepsilon_{\text{in}}\), fuera \(\varepsilon_{\text{out}}\)
  \item Carga total \(Q=\displaystyle\int_{\text{esfera}}\rho(\vec r\,')\,d v'=\frac{4\pi R^{3}}{3}\,\rho\)
\end{itemize}

\paragraph{1) Campos \(\vec D\) y \(\vec E\) por simetría esférica (Ley de Gauss)}
Para \(r<R\) la carga libre encerrada es \(Q_{\text{enc}}(r)=\rho\,\frac{4\pi r^{3}}{3}\). Con \(\oint \vec D\cdot d\vec a=Q_{\text{enc}}\) se obtiene
\[
D_{\text{in}}(r)=\frac{\rho\,r}{3}\,\hat{\boldsymbol r},\qquad
E_{\text{in}}(r)=\frac{D_{\text{in}}}{\varepsilon_{\text{in}}}
=\frac{\rho\,r}{3\,\varepsilon_{\text{in}}}\,\hat{\boldsymbol r}
\]

Para \(r>R\), la carga libre encerrada es \(Q\), de modo que
\[
D_{\text{out}}(r)=\frac{Q}{4\pi r^{2}}\,\hat{\boldsymbol r}
=\frac{\rho R^{3}}{3}\,\frac{1}{r^{2}}\,\hat{\boldsymbol r},\qquad
E_{\text{out}}(r)=\frac{D_{\text{out}}}{\varepsilon_{\text{out}}}
=\frac{\rho R^{3}}{3\,\varepsilon_{\text{out}}}\,\frac{1}{r^{2}}\,\hat{\boldsymbol r}
\]

\paragraph{2) Densidad de energía electrostática}
En un dieléctrico lineal \(w=\tfrac12\,\vec E\cdot\vec D=\tfrac{1}{2}\,\frac{D^{2}}{\varepsilon}\). Por tanto
\[
w_{\text{in}}(r)=\frac{1}{2}\frac{D_{\text{in}}^{2}}{\varepsilon_{\text{in}}}
=\frac{1}{2}\frac{1}{\varepsilon_{\text{in}}}\left(\frac{\rho r}{3}\right)^{2},\qquad
w_{\text{out}}(r)=\frac{1}{2}\frac{D_{\text{out}}^{2}}{\varepsilon_{\text{out}}}
=\frac{1}{2}\frac{1}{\varepsilon_{\text{out}}}\left(\frac{\rho R^{3}}{3r^{2}}\right)^{2}
\]

\paragraph{3) Energía total \(W\) como suma de las contribuciones interior + exterior}
Usamos \(d v=4\pi r^{2}\,dr\)
\[
W=W_{\text{in}}+W_{\text{out}}
=\int_{0}^{R} w_{\text{in}}(r)\,4\pi r^{2}dr
+\int_{R}^{\infty} w_{\text{out}}(r)\,4\pi r^{2}dr
\]

\emph{Parte interior}
\[
W_{\text{in}}=\int_{0}^{R}\frac{1}{2\varepsilon_{\text{in}}}
\left(\frac{\rho r}{3}\right)^{2}4\pi r^{2}\,dr
=\frac{2\pi\rho^{2}}{9\,\varepsilon_{\text{in}}}\int_{0}^{R}r^{4}\,dr
=\frac{2\pi\rho^{2}}{9\,\varepsilon_{\text{in}}}\,\frac{R^{5}}{5}
=\frac{2\pi\rho^{2}R^{5}}{45\,\varepsilon_{\text{in}}}
\]

\emph{Parte exterior}
\[
W_{\text{out}}=\int_{R}^{\infty}\frac{1}{2\varepsilon_{\text{out}}}
\left(\frac{\rho R^{3}}{3r^{2}}\right)^{2}4\pi r^{2}\,dr
=\frac{2\pi\rho^{2}R^{6}}{9\,\varepsilon_{\text{out}}}\int_{R}^{\infty}\frac{dr}{r^{2}}
=\frac{2\pi\rho^{2}R^{6}}{9\,\varepsilon_{\text{out}}}\,\frac{1}{R}
=\frac{2\pi\rho^{2}R^{5}}{9\,\varepsilon_{\text{out}}}
\]

\paragraph{4) Resultado total en función de \(\rho\) y \(R\)}
\[
\boxed{\,W
=\frac{2\pi\rho^{2}R^{5}}{45\,\varepsilon_{\text{in}}}
+\frac{2\pi\rho^{2}R^{5}}{9\,\varepsilon_{\text{out}}}\,}
\]

\paragraph{5) Resultado equivalente en función de la carga total \(Q=\frac{4\pi R^{3}}{3}\rho\)}
Usando \(\rho=\dfrac{3Q}{4\pi R^{3}}\)
\[
\boxed{\,W
=\frac{Q^{2}}{40\pi R\,\varepsilon_{\text{in}}}
+\frac{Q^{2}}{8\pi R\,\varepsilon_{\text{out}}}
=\frac{Q^{2}\big(5\,\varepsilon_{\text{in}}+\varepsilon_{\text{out}}\big)}
{40\pi R\,\varepsilon_{\text{in}}\varepsilon_{\text{out}}}\,}
\]

\paragraph{6) Chequeos y casos límite}
\begin{itemize}
  \item \(\varepsilon_{\text{in}}=\varepsilon_{\text{out}}=\varepsilon\) (mismo material dentro y fuera): 
  \(\displaystyle W=\frac{3}{20}\frac{Q^{2}}{\pi\varepsilon R}\), que coincide con la energía de una esfera con carga uniformemente distribuida en un medio homogéneo
  \item Medio exterior vacío: \(\varepsilon_{\text{out}}=\varepsilon_{0}\), interior con \(\varepsilon_{\text{in}}\) arbitraria, queda la expresión general de arriba
\end{itemize}

\paragraph{7) Comentarios}
\begin{itemize}
  \item Usar \(w=\tfrac12 \vec E\cdot\vec D\) evita contar doble la energía de polarización en medios lineales, pues \(\vec D\) contabiliza la carga libre
  \item Alternativamente, puede obtenerse \(W\) integrando \(\tfrac12 \rho V\) en el volumen de la esfera, con los potenciales interior/exterior consistentes con las condiciones de frontera en \(r=R\)
\end{itemize}
