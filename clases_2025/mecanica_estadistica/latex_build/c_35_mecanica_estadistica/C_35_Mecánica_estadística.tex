\section{Recordando}

\[V_{N}(r) = \frac{r^{N}\pi^{\frac{N}{2}}}{\frac{N}{2}\Gamma \left(\frac{N}{2}\right)}, \quad \text{con} \quad C_{N}= \frac{\pi^{\frac{N}{2}}}{\frac{N}{2}\Gamma \left( \frac{N}{2} \right)}  \]


\[ S_{N}(r) = C_{N}Nr^{N-1}  \]

\[ 2mE=\sum\limits_{i=1}^{3N} p_{i}^{2}  \]

\textbf{Ecuación de la esfera en 3N dim de radio}

\[ \sqrt{2mE} \]


La superficie de la esfea $E=H$ es:


\[ S_{3N}\left( \sqrt{2mE} \right)   \]



Si fijamos el valor de una dimensión $p_{i}=p$ al valor $p$ tenemos:


\[ 2mE-p^{2} = \sum\limits_{i=1}^{3N-1}p_{1}^{2}   \]

Por lo tanto su superficie es proporcional a:


\[ S_{3N-1} \left(  \sqrt{ 2mE-p^{2}  }   \right)   \]

La probabilidad de encontrar al sistema o subsistema con un valor $p$ es:


\[ \rho_{C}(P) = \frac{ S_{3N-1} \left( \sqrt{ 2mE-p^{2} }  \right)dP  }{S_{3N} \left( \sqrt{2mE }  \right) } \]


Veamos:

\begin{equation}
\begin{split}
 S_{3N-1} \left(  \sqrt{2mE-p^{2} }  \right) &= C_{N}N\left( 2mE-p^{2} \right)^{ \frac{N-1}{2} }  \\ 
 &= \frac{ C_{N}N(2mE)^{\frac{N}{2}} }{(2mE-p^{2})^{\frac{1}{2}}} \left( 1 -  \frac{p^{2}}{2mE} \right)^{\frac{N}{2}} 
\end{split}
\end{equation}



En el límite termodinámico:

\[ N \to \infty; E \to \infty \quad \text{de modo que} \quad \to \frac{E}{N} \to e, \quad \text{finito}\]


Tenemos:


\[ \rho_{c}(p) \longrightarrow C_{0} \left(1- \frac{p^{2}}{2m\underbrace{\frac{E}{N}}_{e}N} \right)  \]

\[\to C_{0}\left( e^{\frac{-p^{2}}{2me}}\right)^{\frac{3}{2}}\]


Tenemos:



\[ \rho_{c}(p) \underbrace{\overbrace{\longrightarrow}^{N \to \infty}}_{E\to \infty} C_{0}e^{- \frac{3}{2}\frac{p^{2}}{2me}}  \]

Como:


\[ \boxed{e=\frac{3}{2} K_{B}T } \]

Entonces:


\[  \rho_{c}(p)  \underbrace{\overbrace{\longrightarrow}^{N \to \infty}}_{E\to \infty} C_{0}e^{ - \frac{1}{K_{B}T} \left( \frac{p^{2}}{2m}\right) } \]

Tenemos:

\[ \epsilon_{n} = \hbar w \left( n+\frac{1}{2}   \right); \quad n=0,1,2     \]

Sabemos que:

\[ Z(T,V,N) = \left[ Z(T,V,1)  \right]^{N} \]

Además:


\[ Z(T,V,1) = \sum\limits_{n=0}^{\infty} e^{-\beta\epsilon_{n}}   \]


Al sustituir la expresión $\epsilon_{n}$

Tenemos:

\begin{align}
Z(T,V,1) &= \sum_{n=0}^{\infty} e^{-\beta \hbar \omega\left(n+\tfrac{1}{2}\right)} \\
&= e^{-\beta \hbar \omega/2}\sum_{n=0}^{\infty} \left(e^{-\beta \hbar \omega}\right)^n \\
&= \frac{e^{-\beta \hbar \omega/2}}{1 - e^{-\beta \hbar \omega}}
= \frac{1}{2\sinh\!\left(\tfrac{\beta \hbar \omega}{2}\right)}
\end{align}


% Partición de un oscilador
\[
Z(T,V,1)=\frac{e^{-\beta \hbar \omega/2}}{1-e^{-\beta \hbar \omega}}
\]

% Energía libre para N osciladores no interactuantes
\[
F(T,V,N) = -N k_B T \ln Z(T,V,1)
= N\frac{\hbar \omega}{2} + N k_B T \ln\!\big(1 - e^{-\beta \hbar \omega}\big)
\]

\[  Z(T,V,1) = \frac{1}{ e^{ \frac{\beta \hbar w}{2}}  - e^{ \frac{-\beta \hbar w}{2}}  } = \frac{1}{  2\sen{h}\left(  \frac{\beta \hbar w}{2}  \right)  } \]
