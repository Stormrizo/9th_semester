\section{Próximo examen: problema de Kepler (dos cuerpos)}

\subsection*{Qué dominar}
\begin{itemize}
  \item Identificar el \textbf{potencial kepleriano}: \(V(r)=-\dfrac{\kappa}{r}\), con \(\kappa=G\,m_1 m_2\).
  \item Escribir el \textbf{lagrangiano} y separar el \textbf{movimiento del centro de masa} y el \textbf{movimiento relativo}.
  \item Derivar las \textbf{ecuaciones de movimiento} para una \textbf{fuerza central} y sus consecuencias: conservación de \(\vec{L}\), movimiento planar y forma de la órbita.
\end{itemize}

\subsection*{Lagrangiano y separación CM/relativo}
Sea
\[
M=m_1+m_2,\qquad \mu=\frac{m_1 m_2}{m_1+m_2},\qquad \kappa=G\,m_1 m_2.
\]
Con \(\vec{R}=\dfrac{m_1\vec{r}_1+m_2\vec{r}_2}{M}\) (centro de masa) y \(\vec{r}=\vec{r}_2-\vec{r}_1\) (coordenada relativa), el lagrangiano se separa como
\[
L=\underbrace{\frac{1}{2}M\,\dot{\vec{R}}^{\,2}}_{\text{coordenadas mayúsculas }(CM)}
\;+\;
\underbrace{\Big(\frac{1}{2}\mu\,\dot{\vec{r}}^{\,2}+\frac{\kappa}{r}\Big)}_{\text{coordenadas minúsculas }(\text{relativo})}.
\]
Es decir,
\[
L=L_{\vec{R}}+L_{\vec{r}},\qquad
L_{\vec{R}}=\frac{1}{2}M\,\dot{\vec{R}}^{\,2},\qquad
L_{\vec{r}}=\frac{1}{2}\mu\,\dot{\vec{r}}^{\,2}+\frac{\kappa}{r}\quad (V=-\kappa/r).
\]

\subsection*{Centro de masa}
Del principio de inercia del CM:
\[
\frac{d}{dt}\big(M\dot{\vec{R}}\big)=\vec{0}
\;\;\Rightarrow\;\;
\vec{R}(t)=\vec{R}_{0}+\vec{V}_{0}\,t
\quad\text{(MRU).}
\]

\subsection*{Movimiento relativo y fuerza central}
Para el movimiento relativo,
\[
\mu\,\ddot{\vec{r}}=-\frac{\partial V}{\partial r}\,\hat{\mathbf r}=-\frac{\kappa}{r^2}\,\hat{\mathbf r}
\;\;\Leftrightarrow\;\;
\mu\,\ddot{\vec{r}}=-\kappa\,\frac{\vec{r}}{r^3}.
\]
Al ser una \textbf{fuerza central} \(\vec{F}=f(r)\,\hat{\mathbf r}\), el \textbf{momento angular orbital} se conserva:
\[
\vec{L}=\mu\,\vec{r}\times\dot{\vec{r}}=\text{cte},\qquad
\dot{\vec{L}}=\vec{r}\times\vec{F}=\vec{0}.
\]
Esto fija un \textbf{plano de movimiento} perpendicular a \(\vec{L}\). La “sabor esférico” del problema proviene de la simetría radial del potencial (depende solo de \(r\)), pero la dinámica efectiva vive en un plano con coordenadas polares \((r,\phi)\).

\subsection*{Geometría angular y ortogonalidad}
Como \(\vec{L}\) es constante, podemos tomar un marco inercial con \(\vec{L}=\ell\,\hat{\mathbf z}\). Se cumple
\[
\vec{L}\perp\vec{r},\qquad \vec{L}\perp\vec{p}=\mu\,\dot{\vec{r}}.
\]
En el plano orbital:
\[
\ell=\mu\,r^2\,\dot{\phi},\qquad
E=\frac{1}{2}\mu\big(\dot r^{\,2}+r^2\dot{\phi}^{\,2}\big)-\frac{\kappa}{r}.
\]
Si \(|\vec{r}\times\vec{p}|=0\) entonces \(r\,p\,\sin\angle(\hat{\mathbf r},\hat{\mathbf p})=0\), lo que implica velocidad puramente radial (caso \(L=0\)).

\subsection{Caso con plano $z=0$}
Sin pérdida de generalidad, en el plano orbital:
\[
\theta=\frac{\pi}{2},\qquad \dot{\theta}=0,\qquad \vec{L}=\ell\,\hat{\mathbf z}.
\]

\subsection{Lagrangiana útil a recordar (coordenadas polares en el plano)}
\[
L=\frac{1}{2}\mu\big(\dot r^{2}+r^{2}\dot{\phi}^{2}\big)+\frac{\kappa}{r},\qquad
\ell=\mu r^{2}\dot{\phi},\qquad
E=\frac{1}{2}\mu\big(\dot r^{2}+r^{2}\dot{\phi}^{2}\big)-\frac{\kappa}{r}.
\]

\subsection*{Órbitas cónicas y ecuación polar}
La solución general (leyes de Kepler) es una \textbf{cónica}:
\[
r(\phi)=\frac{p}{1+e\cos(\phi-\bar{\phi})},
\qquad
p=\frac{\ell^{2}}{\mu\,\kappa},\qquad
e=\sqrt{1+\frac{2E\ell^{2}}{\mu\,\kappa^{2}}}\ \ge 0.
\]
Equivalente en la forma que usas:
\[
\frac{1}{r}=\frac{\mu\kappa}{\ell^{2}}
\left[
1+\sqrt{1+\frac{2E\ell^{2}}{\mu\kappa^{2}}}\,\cos(\phi-\bar{\phi})
\right].
\]
Clasificación por energía: \(E<0\) (elipse), \(E=0\) (parábola), \(E>0\) (hipérbola).

\subsection{Caso para determinar la geometría y el marco de referencia}

Para fijar la \textbf{geometría} (plano, ángulos y rotaciones) necesitamos:
\begin{itemize}
  \item \textbf{Condiciones iniciales} respecto a un marco de referencia inercial.
  \item El \textbf{momento angular orbital} \(\vec{L}\), que fija el plano de la órbita.
\end{itemize}

En el caso planar estándar, tomamos el \textbf{plano orbital} como \(z=0\), de modo que \(\vec{r}\in\mathbb{R}^2\) y \(\vec{L}=\ell\,\hat{\mathbf{z}}\).

\medskip

\noindent\textbf{Ángulo de orientación de la cónica.}  
La orientación de la cónica en el plano se describe por un ángulo de referencia:
\[
\phi \quad \text{o bien} \quad \bar{\phi},
\]
donde \(\bar{\phi}\) suele representar un \emph{desfase angular} (por ejemplo, ángulo del periastro respecto al eje \(x\) en la figura). En estas notas, \(\bar{\phi}\) es el ángulo que barre el eje mayor respecto al eje \(x\).

\noindent\textbf{Ecuación polar de la cónica en el plano \(z=0\).}  
Conservando tu notación (\(m\) masa reducida, \(\ell\) momento angular escalar, \(E\) energía), la órbita se escribe como
\[
\frac{1}{r} \;=\; \frac{m\,k}{\ell^{2}} \left[
  1 + \sqrt{\,1+\frac{2E\,\ell^{2}}{m\,k^{2}}\,}\,
  \cos\!\big(\phi - \bar{\phi}\big)
\right],
\]
donde el término bajo la raíz determina la \textbf{excentricidad} (\(E<0\Rightarrow\) elipse, \(E=0\Rightarrow\) parábola, \(E>0\Rightarrow\) hipérbola).  
\emph{Si la energía \(E\) cambia, cambia el tipo de trayectoria.}

\medskip

\noindent\textbf{Vector de posición en el plano orbital.}  
En el plano \(z=0\), el vector de posición es
\[
\vec{r}_{z=0} \;=\; r\big(\cos\phi\,\hat{\mathbf{i}}+\sin\phi\,\hat{\mathbf{j}}\big).
\]

\noindent\textbf{Rotación del sistema (cambio de base/orientación).}  
Si la cónica necesita rotarse un ángulo \(\alpha\) alrededor de \(\hat{\mathbf{z}}\) (por ejemplo, para alinear el periastro con el eje \(x\) o para introducir \(\bar{\phi}\)), usamos la matriz de rotación en \(2\)D o en \(3\)D:
\[
R_z(\alpha) \;=\;
\begin{pmatrix}
\cos\alpha & -\sin\alpha & 0\\
\sin\alpha & \phantom{-}\cos\alpha & 0\\
0 & 0 & 1
\end{pmatrix},
\qquad
\vec{r}_{np} \;=\; R_z(\alpha)\,\vec{r}.
\]
En tu notación compacta:
\[
\vec{r}_{np} \;=\; \big(R\,\vec{r}\big),
\]
donde \(R\) es la rotación adecuada (p. ej., \(R_z(\bar{\phi})\) para desplazar la cónica por \(\bar{\phi}\)).  
Si más adelante el plano orbital no coincidiera con \(z=0\), puede emplearse una \emph{secuencia de Euler} \((\Omega,i,\omega)\), pero para este curso basta con la rotación en el plano.

\medskip

\noindent\textbf{Descomposición CM/relativo (para ubicar cada cuerpo).}
\[
\vec{R} \;=\; \frac{ m_{1}\,\vec{r}_{1} + m_{2}\,\vec{r}_{2} }{ m_{1}+m_{2} },
\qquad
\vec{r} \;=\; \vec{r}_{2}-\vec{r}_{1}.
\]
A partir de esto, las posiciones de cada cuerpo en función de \(\vec{R}\) y \(\vec{r}\) son

\[  \vec{r}_{1} = \vec{r}_{1}(\vec{R}, \vec{r}) \]

\[ \vec{r}_{2} = \vec{r}_{2} (\vec{R}, \vec{r})  \]


\[
\boxed{
\ \vec{r}_{1} \;=\; \vec{R} \;-\; \frac{m_{2}}{m_{1}+m_{2}}\,\vec{r},
\qquad
\vec{r}_{2} \;=\; \vec{R} \;+\; \frac{m_{1}}{m_{1}+m_{2}}\,\vec{r}\ }.
\]
Estas relaciones son las que luego se \emph{rotan} con \(R\) si deseas expresar la órbita en un marco girado.

\noindent\textbf{Resumen operativo para elegir geometría:}
\begin{enumerate}
  \item Fija el \textbf{plano orbital} con \(\vec{L}\) (tomar \(z=0\) es natural si \(\vec{L}\parallel\hat{\mathbf{z}}\)).
  \item Usa la ecuación polar \(r(\phi)\) y el \textbf{desfase} \(\bar{\phi}\) para orientar la cónica.
  \item Construye \(\vec{r}_{z=0}\) y, si hace falta, aplica \(\vec{r}_{np}=R_z(\alpha)\vec{r}\).
  \item Recupera \(\vec{r}_{1},\vec{r}_{2}\) con \(\vec{R}\) y \(\vec{r}\) (y rota si el marco lo exige).
\end{enumerate}
